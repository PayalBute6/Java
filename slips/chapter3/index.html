AccountDetail.java

Question 1: Write a program to define a class Account having members custname, accno. Define default and parameterized constructor. Create a subclass called SavingAccount with members savingbal, minbal. Create a derived class AccountDetail that extends the class SavingAccount with members, depositamt and withdrawalamt. Write a appropriate method to display customer details.
// Base Class
class Account {
    String customerName;
    int accountNumber;

    // Default constructor
    public Account() {
        customerName = "Unknown";
        accountNumber = 0;
    }

    // Parameterized constructor
    public Account(String name, int number) {
        customerName = name;
        accountNumber = number;
    }
}

// Subclass 1
class SavingAccount extends Account {
    double savingBalance;
    double minimumBalance;

    // Default constructor
    public SavingAccount() {
        super(); 
        savingBalance = 0.0;
        minimumBalance = 0.0;
    }

    // Parameterized constructor
    public SavingAccount(String name, int number, double balance, double minBalance) {
        super(name, number); 
        savingBalance = balance;
        minimumBalance = minBalance;
    }
}

// Subclass 2 (Final derived class)
public class AccountDetail extends SavingAccount {
    double depositAmount;
    double withdrawalAmount;

    // Default constructor
    public AccountDetail() {
        super(); 
        depositAmount = 0.0;
        withdrawalAmount = 0.0;
    }

    // Parameterized constructor
    public AccountDetail(String name, int number, double balance, double minBalance, double deposit, double withdraw) {
        super(name, number, balance, minBalance); 
        depositAmount = deposit;
        withdrawalAmount = withdraw;
        
        // Calculate the final balance
        savingBalance = savingBalance + depositAmount - withdrawalAmount;
    }

    // Display details
    public void displayDetails() {
        System.out.println("Name: " + customerName);
        System.out.println("Account No: " + accountNumber);
        System.out.println("Current Balance: Rs. " + savingBalance);
    }

    public static void main(String[] args) {
        // Create customer object
        AccountDetail cust = new AccountDetail("Radha", 101, 5000.0, 1000.0, 2000.0, 1500.0);
        
        // Display details
        cust.displayDetails();
    }
}

Output:

> javac AccountDetail.java
> java AccountDetail
Name: Radha
Account No: 101
Current Balance: Rs. 5500.0

PlaceDemo.java

Question 2: Write a program for multilevel inheritance such that Country is inherited from Continent. State is inherited from Country. Display the place, State, Country and Continent.
// Level 1
class Continent {
    String continentName;

    public Continent(String name) {
        continentName = name;
    }
}

// Level 2
class Country extends Continent {
    String countryName;

    public Country(String continent, String country) {
        super(continent); 
        countryName = country;
    }
}

// Level 3
class State extends Country {
    String stateName;

    public State(String continent, String country, String state) {
        super(continent, country);
        stateName = state;
    }
}

// Level 4
class Place extends State {
    String placeName;

    public Place(String continent, String country, String state, String place) {
        super(continent, country, state);
        placeName = place;
    }

    public void displayLocation() {
        System.out.println("Continent: " + continentName);
        System.out.println("Country: " + countryName);
        System.out.println("State: " + stateName);
        System.out.println("Place: " + placeName);
    }
}

public class PlaceDemo {
    public static void main(String[] args) {
        Place myPlace = new Place("Asia", "India", "Maharashtra", "Pune");
        myPlace.displayLocation();
    }
}

Output:

> javac PlaceDemo.java
> java PlaceDemo
Continent: Asia
Country: India
State: Maharashtra
Place: Pune

ShapeDemo.java

Question 4: Write a program to create an abstract class named Shape that contains two integers and an empty method named printArea(). Provide three classes named Rectangle, Triangle and Circle such that each one of the classes extends the class Shape. Each one of the classes contain only the method printArea() that prints the area of the given shape. (use method overriding).
// Abstract Class
abstract class Shape {
    int dim1;
    int dim2;

    abstract void printArea();
}

// Rectangle
class Rectangle extends Shape {
    public Rectangle(int length, int width) {
        dim1 = length;
        dim2 = width;
    }

    void printArea() {
        int area = dim1 * dim2;
        System.out.println("Area of Rectangle is: " + area);
    }
}

// Triangle
class Triangle extends Shape {
    public Triangle(int base, int height) {
        dim1 = base;
        dim2 = height;
    }

    void printArea() {
        double area = 0.5 * dim1 * dim2;
        System.out.println("Area of Triangle is: " + area);
    }
}

// Circle
class Circle extends Shape {
    public Circle(int radius) {
        dim1 = radius;
        dim2 = 0; // Not used
    }

    void printArea() {
        double area = 3.14 * dim1 * dim1;
        System.out.println("Area of Circle is: " + area);
    }
}

public class ShapeDemo {
    public static void main(String[] args) {
        Shape rect = new Rectangle(10, 5);
        rect.printArea();

        Shape tri = new Triangle(10, 8);
        tri.printArea();

        Shape circ = new Circle(7);
        circ.printArea();
    }
}

Output:

> javac ShapeDemo.java
> java ShapeDemo
Area of Rectangle is: 50
Area of Triangle is: 40.0
Area of Circle is: 153.86

SimpleCalcDemo.java

Question 5: Write a Java program that defines an interface Calculator containing methods add() and subtract() and implements it in class SimpleCalc.
// Interface
interface Calculator {
    int add(int a, int b);
    int subtract(int a, int b);
}

// Class implementing interface
class SimpleCalc implements Calculator {
    
    public int add(int a, int b) {
        return a + b;
    }

    public int subtract(int a, int b) {
        return a - b;
    }
}

public class SimpleCalcDemo {
    public static void main(String[] args) {
        SimpleCalc calc = new SimpleCalc();

        int sum = calc.add(50, 15);
        int diff = calc.subtract(50, 15);

        System.out.println("Addition: " + sum);
        System.out.println("Subtraction: " + diff);
    }
}

Output:

> javac SimpleCalcDemo.java
> java SimpleCalcDemo
Addition: 65
Subtraction: 35
