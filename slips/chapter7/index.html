ChangeHeadingText.html

Question 1: Create a webpage containing a heading and a button. When the button is clicked, change the heading text from “Hello! Welcome” to “Text Changed Successfully!”.
<!DOCTYPE html>
<html>
<head>
    <title>Change Heading Text</title>

</head>
<body>

    <!-- 1. The heading. We give it an 'id' so JavaScript can find it easily -->
    <h2 id="myHeading">Hello! Welcome</h2>

    <!-- 2. The button. The 'onclick' attribute tells it to run our JavaScript function when clicked -->
    <button onclick="changeText()">Click me to change the text</button>

    <!-- 3. The JavaScript code -->
    <script>
        // This function is triggered when the button is clicked
        function changeText() {
            
            // Step 1: Find the heading using its id ("myHeading")
            // Step 2: Change the text inside it using .innerText
            document.getElementById("myHeading").innerText = "Text Changed Successfully!";
            
        }
    </script>

</body>
</html>

ChangeImage.html

Question 2: Create a webpage using JavaScript DOM manipulation and event handling with an image and a button. When the button is clicked, replace the displayed image with another image.
<!DOCTYPE html>
<html>
<head>
    <title>Change Image with JavaScript</title>
    <style>
        img {
            width: 400px;
            height: 300px;
            display: block;
            margin-bottom: 10px;
        }
    </style>
</head>
<body>

    <h2>JavaScript Image Replacer</h2>

    <!-- 1. The image. We give it an 'id' so JavaScript can find it easily -->
    <!-- It starts by loading a random forest image from picsum.photos -->
    <img id="myImage" src="https://picsum.photos/id/10/400/300" alt="First Image">

    <!-- 2. The button. The 'onclick' event handler tells it to run our function when clicked -->
    <button onclick="changeImage()">Click to Change Image</button>

    <!-- 3. The JavaScript code (DOM Manipulation) -->
    <script>
        // This function runs when the button is clicked
        function changeImage() {
            
            // Step 1: Find the image element using its id ("myImage")
            var imgElement = document.getElementById("myImage");
            
            // Step 2: Replace the 'src' (source) attribute with a brand new image link
            imgElement.src = "https://picsum.photos/id/11/400/300";
            
        }
    </script>

</body>
</html>
